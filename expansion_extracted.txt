// EXPANSION — comprehensive enhancement layer (2026-10-28 directive).
// Additive only: the verified V2 bank, the Ammo bank, and every master
// stay word-for-word untouched. This module adds:
//  1. NEW LANES — emergent / regional pockets plus the strongest Ammo
//     styles promoted to full lanes. Genre substance is drawn from the
//     two hip-hop production reference documents in the beat goal's
//     sources (pasted 2026-10-27, unverified quarry material); anything
//     not in those sources is plain generic production description.
//     No artist or producer names anywhere. No Suno facts are claimed.
//  2. STRUCTURAL FRAMEWORKS — a named profile per lane (old and new):
//     its signature layout, where its builds / drops / breakdowns sit,
//     and its signature events. The lyrics-map generator, the Simple
//     composer, and the remix templates all read from these profiles,
//     so structures differ by genre instead of sharing one skeleton.
//  3. ANALYZER AFFINITY — per-lane tempo / brightness / density /
//     low-end affinities so Analyze can rank lane suggestions across
//     the whole library, old and new lanes alike.

import { LANES, LOCK_LINE, type JobType } from "./bank";

export { LOCK_LINE };

export interface FrameworkProfile {
  /** Named framework, e.g. "Low-End Bounce Frame". */
  name: string;
  /** Signature bar layout in one line. */
  layout: string;
  /** Where / how the build-ups sit (short clause, map-ready). */
  build: string;
  /** Where / how the drops land (short clause, map-ready). */
  drop: string;
  /** Where / how the breakdown sits (short clause, map-ready). */
  breakdown: string;
  /** Signature events, comma list. */
  signature: string;
}

export function isExpansionLane(lane: string): boolean {

// ---------------------------------------------------------------------------
// EXPANSION STYLE PROMPTS — 2 Original + 2 Preserve per new lane, each
// ≤200 chars, no artist names; Preserve carries the exact lock line.
// Composed from the lane's own texture + drum signature.
// ---------------------------------------------------------------------------

export interface ExpansionStylePrompt {
  lane: string;
  job: JobType;
  text: string;
}

function cap(s: string): string {
  return s.charAt(0).toUpperCase() + s.slice(1);
}

export const EXPANSION_STYLE_PROMPTS: ExpansionStylePrompt[] = (() => {
  const out: ExpansionStylePrompt[] = [];
  for (const l of EXPANSION_LANES) {
    out.push({
      lane: l.name,
      job: "original",
      text: `${cap(l.name)} instrumental, ${l.texture}, ${l.short}.`,
    });
    out.push({
      lane: l.name,
      job: "original",
      text: `${cap(l.name)} pocket, ${l.texture}, ${l.short}, fills at the turns, chorus lifts.`,
    });
    out.push({
      lane: l.name,
      job: "preserve",
      text: `${LOCK_LINE}. ${cap(l.name)} canvas: ${l.short}, spice only, sample stays lead.`,
    });
    out.push({
      lane: l.name,
      job: "preserve",
      text: `${LOCK_LINE}. Add ${l.name} drums: ${l.short}, spice only, sample stays lead.`,
    });
  }
  return out;
})();

export function expansionStylePromptsFor(
  job: JobType,
  lane: string,
): ExpansionStylePrompt[] {
  return EXPANSION_STYLE_PROMPTS.filter((p) => p.job === job && p.lane === lane);
}

export function expansionDrums(lane: string): string | undefined {
  return expansionLane(lane)?.drums;
}

export interface LaneRank {
  lane: string;
  score: number;
  why: string;
}

export function rankLanesForBeat(
  bpm: number | null,
  centroidHz: number,
  onsetRate: number,
  limit = 3,
): LaneRank[] {
  const brightBand = centroidHz >= 3400 ? "bright" : centroidHz >= 2200 ? "mid" : "dark";
  const densityBand = onsetRate >= 3 ? "busy" : onsetRate >= 1.6 ? "steady" : "sparse";
  const scored: LaneRank[] = [];
  for (const lane of ALL_LANES) {
    const exp = expansionLane(lane);
    const aff: Affinity | undefined = exp
      ? { bpm: exp.bpm, bright: exp.bright, density: exp.density }
      : OLD_AFFINITY[lane];
    if (!aff) continue;
    let score = 0;
    const why: string[] = [];
    if (bpm !== null) {
      const [lo, hi] = aff.bpm;
      if (bpm >= lo && bpm <= hi) {
        score += 3;
        why.push(`tempo sits in its ${lo}–${hi} pocket`);
      } else if (bpm >= lo / 2 && bpm <= hi / 2 + 8) {
        score += 1;
        why.push("tempo fits at half-time");
      }
    }
    if (aff.bright === "any" || aff.bright === brightBand) {
      score += 1;
      if (aff.bright !== "any") why.push(`${brightBand} top end matches`);
    }
    if (aff.density === "any" || aff.density === densityBand) {
      score += 1;
      if (aff.density !== "any") why.push(`${densityBand} hit density matches`);
    }
    scored.push({
      lane,
      score,
      why: why.length > 0 ? why.join(" · ") : "nearest available pocket",
    });
  }
  scored.sort((a, b) => b.score - a.score);
  return scored.slice(0, limit);
}
