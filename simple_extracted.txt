// SIMPLE bank — prompts written for Suno's Simple mode: ONE text box
// (the screenshot Boo sent: Simple mode, a single prompt field, +Audio
// upload, v6). Simple mode has no separate Style / Exclude / Lyrics
// boxes, so each Simple prompt is a single self-contained paragraph in
// plain language that does the whole job in that one field: genre and
// pocket, drum and 808 direction, and the full structure spelled out in
// words (intro, verses, choruses, builds, drops, fills).
//
// This bank is ADDITIVE: every prompt is composed at load time from
// the existing verified V2 bank and the Ammo bank — those banks' own
// text is never altered, merged, or overwritten. Preserve forms always
// carry the exact lock line inside the paragraph. The Simple prompts
// are NOT fenced off in a silo (Boo's standing approach is additive):
// they live in their own Simple section AND are woven into the main
// Library and the Build flow alongside everything else. Only complete
// duplicates are skipped.
//
// No character cap is claimed for the Simple box: the 2026-10-27
// official-source audit found no published Simple-box cap, so the UI
// shows character counts as guidance only, never as a Suno limit.

 * Community working size for the Simple-mode prompt: ~3,000 chars,
 * from maintained Suno v6 best-practice references (Suno itself

 * never a claimed Suno limit.

const STRUCTURE_ORIGINAL =
  "Structure, in words: intro of 8 bars with melody and tops only and a reverse swell in, verse 1 for 16 bars, a build-up with a riser, snare roll and FX sweep into chorus 1 for 8 bars with a heavy drop on bar 1, a post-chorus drop and a stop-time cut, verse 2 for 16 bars with a mid-verse drop-out to melody and tops only then a drop back in on the one, a second build-up into chorus 2 for 8 bars, verse 3 for 16 bars with drum fills closing every 4 bars, a breakdown to melody and tops only, a final build-up at maximum tension, a final chorus for 8 bars with the heaviest drop and a half-time finish, a final drop, and an outro that strips back layer by layer to a tape-stop finish. Fills close the sections, builds rise into every chorus, and cuts and rolls mark the transitions.";

const STRUCTURE_PRESERVE =
  "Structure, in words: intro with the uploaded sample and sampled vocals playing solo exactly as performed, verse 1 for 16 bars with canvas drums only underneath, a build-up with a riser and snare roll into chorus 1 for 8 bars with a heavy canvas drop on bar 1, a post-chorus drop and a stop-time cut in the canvas only, verse 2 for 16 bars with a mid-verse drop-out back to the sample solo then a drop back in on the one, a second build-up into chorus 2 for 8 bars, verse 3 for 16 bars with canvas fills closing every 4 bars, a breakdown to the sample and tops only, a final build-up, a final chorus for 8 bars with the heaviest canvas drop and a half-time finish, a final drop, and an outro that strips the canvas back so the sample plays out alone to a tape-stop finish. Every fill, build, drop, cut, roll and FX move happens in the canvas drums only — no new lead melody, no new or re-sung vocals.";

const GENERIC_DRUMS =
  "pocket locked tight, 808 tuned to the key and kept under the lead, fills short at the section turns, space between the hits";

/**
 * Compose one Simple-box paragraph from a base prompt drawn from the
 * verified bank or the Ammo bank. The base text is used verbatim as the
 * opening; the drum/808 direction and the worded structure are appended
 * so the single field carries everything Simple mode needs.
 */
/** Worded structure for a lane — reads from the lane's structural
 * framework profile so builds, drops and the breakdown sit where
 * that genre puts them, instead of one shared skeleton. */
export function simpleStructureFor(lane: string, job: JobType): string {
  const base = job === "preserve" ? STRUCTURE_PRESERVE : STRUCTURE_ORIGINAL;
  const fw = frameworkFor(lane);
  if (!fw) return base;
  return `${base} Framework — ${fw.name} (${fw.layout}): builds — ${fw.build}; drops — ${fw.drop}; breakdown — ${fw.breakdown}; signature events — ${fw.signature}.`;
}

export function composeSimple(base: string, lane: string, job: JobType): string {
  let head = base.trim().replace(/\s+/g, " ");
  if (job === "preserve" && !head.includes(LOCK_LINE)) {
    head = `${LOCK_LINE}. ${head}`;
  }
  if (head && !/[.!?]$/.test(head)) head += ".";
  const drums = laneDrums(lane);
  const drumSentence = `Drums and 808: ${drums === laneDrums("__none__") ? GENERIC_DRUMS : drums}.`;
  const structure = simpleStructureFor(lane, job);
  return `${head} ${drumSentence} ${structure}`.replace(/\s+/g, " ").trim();
}

export interface SimplePrompt {
  key: string;
  /** Lane name (verified bank) or Ammo section name. */
  lane: string;
  title: string;
  /** "Lane bank" | "Gospel set" | "Ammo" — where the base came from. */
  source: string;
  job: JobType;
  text: string;
}

export const SIMPLE_PROMPTS: SimplePrompt[] = (() => {
  const raw: SimplePrompt[] = [];
  const items = raw;
  // The 21 lanes, from the verified V2 bank (Track 1 Original +
  // Track 2 Preservation style prompts). Track 3 utility add-ons are
  // fragments, not full beats, so they are not turned into Simple
  // prompts.
  bank.style_prompts.forEach((p, i) => {
    if (p.track !== 1 && p.track !== 2) return;
    const job: JobType = p.track === 2 ? "preserve" : "original";
    items.push({
      key: `simple-lane-${i}`,
      lane: p.subgenre,
      title: p.subgenre,
      source: "Lane bank",
      job,
      text: composeSimple(p.text, p.subgenre, job),
    });
  });
  // Expansion lanes — the new subgenre lanes (additive layer in
  // expansion.ts), composed the same way from their own style prompts.
  EXPANSION_STYLE_PROMPTS.forEach((p, i) => {
    items.push({
      key: `simple-expansion-${i}`,
      lane: p.lane,
      title: p.lane,
      source: "Expansion lane",
      job: p.job,
      text: composeSimple(p.text, p.lane, p.job),
    });
  });
  // Gospel set — preserve-only in the bank; the compose step guarantees
  // the exact lock line leads every one of these paragraphs.
  bank.gospel_prompts.forEach((p, i) => {
    items.push({
      key: `simple-gospel-${i}`,
      lane: "Gospel trap",
      title: "Gospel trap",
      source: `Gospel set · ${p.group}`,
      job: "preserve",
      text: composeSimple(p.text, "Gospel trap", "preserve"),
    });
  });
  // Ammo styles — 94 styles × Original/Preserve, kept labeled as the
  // separate Ammo bank, same as everywhere else in the app.
  ammo.sections.forEach((sec) => {
    sec.styles.forEach((st) => {
      (["original", "preserve"] as const).forEach((job) => {
        items.push({
          key: `simple-ammo-${sec.section}-${st.name}-${job}`,
          lane: sec.section,
          title: st.name,
          source: "Ammo",
          job,
          text: composeSimple(ammoText(st, job), sec.section, job),
        });
      });
    });
  });
  // Skip only complete duplicates — same full paragraph already in.
  const seen = new Set<string>();
  return raw.filter((p) => {
    if (seen.has(p.text)) return false;
    seen.add(p.text);
    return true;
  });
})();

/** Distinct filter values for the Simple browse list, in bank order. */
export const SIMPLE_LANES: string[] = (() => {
  const seen: string[] = [];
  for (const p of SIMPLE_PROMPTS) {
    if (!seen.includes(p.lane)) seen.push(p.lane);
  }
  return seen;
})();

export function simplePromptsFor(job: JobType, lane: string): SimplePrompt[] {
  return SIMPLE_PROMPTS.filter((p) => p.job === job && p.lane === lane);
}

/** First Simple prompt for a lane/job — the Build screen's default. */
export function firstSimple(job: JobType, lane: string): SimplePrompt | undefined {
  return SIMPLE_PROMPTS.find((p) => p.job === job && p.lane === lane);
}

/** A random Simple prompt, optionally pinned to a job and/or lane. */
export function randomSimple(job?: JobType, lane?: string): SimplePrompt | undefined {
  const pool = SIMPLE_PROMPTS.filter(
    (p) =>
      (job === undefined || p.job === job) &&
      (lane === undefined || p.lane === lane),
  );
  if (pool.length === 0) return undefined;
  return pool[Math.floor(Math.random() * pool.length)];
}

/** The worded structure sentence for a job — the third layer of an
 * assembled Simple package, and the tail of every full paragraph. */
export function simpleStructure(job: JobType): string {
  return job === "preserve" ? STRUCTURE_PRESERVE : STRUCTURE_ORIGINAL;
}

// TRANSFORMATION DIRECTIVES — the second Simple shape, mirroring how
// the Simple box actually gets used (Boo's screenshots, 2026-10-28):
// the box sits on an uploaded audio and its "+" menu inserts existing
// content (Lyrics, Styles, Playlist, Image, Video, Audio, Voice), so
// the prompt itself is a SHORT transformation directive in the shape
// of Suno's own example — "Turn this into the lofi hip hop version".
// One per lane, Original and Preserve; Preserve carries the exact lock
// line. These pair with an inserted style and the worded structure in
// the package assembler (assembleSimplePackage below).
export const SIMPLE_DIRECTIVES: SimplePrompt[] = (() => {
  const items: SimplePrompt[] = [];
  for (const lane of ALL_LANES) {
    const drums = laneDrums(lane);
    items.push({
      key: `simple-directive-${lane}-original`,
      lane,
      title: lane,
      source: "Directive",
      job: "original",
      text: `Turn this into the ${lane} version — ${drums}, verses stay in the pocket, choruses lift.`,
    });
    items.push({
      key: `simple-directive-${lane}-preserve`,
      lane,
      title: lane,
      source: "Directive",
      job: "preserve",
      text: `${LOCK_LINE}. Turn this upload into the ${lane} version — ${drums}, spice only, sample stays lead.`,
    });
  }
  return items;
})();

export function directiveFor(job: JobType, lane: string): SimplePrompt | undefined {
  return SIMPLE_DIRECTIVES.find((d) => d.job === job && d.lane === lane);
}

/**
 * Assemble a Simple package the way the box is actually filled:
 * transformation directive + an inserted bank style prompt (the "+"
 * menu's Styles insert) + the worded structure, as one copy-ready
 * paragraph. The lock line appears once: if both the directive and
 * the inserted Preserve style carry it, the style's copy is dropped.
 */
export function assembleSimplePackage(
  directive: string,
  styleText: string,
  job: JobType,
): string {
  const head = directive.trim().replace(/\s+/g, " ");
  let style = styleText.trim().replace(/\s+/g, " ");
  if (job === "preserve" && head.includes(LOCK_LINE)) {
    const prefix = `${LOCK_LINE}. `;
    if (style.startsWith(prefix)) style = style.slice(prefix.length);
    else if (style.startsWith(`${LOCK_LINE} — `))
      style = style.slice(LOCK_LINE.length + 3);
  }
  const parts = [head, style, simpleStructure(job)].filter(Boolean);
  return parts.join(" ").replace(/\s+/g, " ").trim();
}

function composeRemix(
  displayName: string,
  drumsLane: string,
  family: RemixFamilyDef,
  job: JobType,
): string {
  const drums = laneDrums(drumsLane);
  const opening = family.open(displayName);
  const fw = frameworkFor(displayName);
  const sig = fw ? ` Signature events: ${fw.signature}; drops — ${fw.drop}.` : "";
  if (job === "preserve") {
    const flavor = family.flavorPreserve(drums);
    return `${LOCK_LINE}. ${opening} — ${flavor}. ${family.structPreserve}${sig}`
      .replace(/\s+/g, " ")
      .trim();
  }
  const flavor = family.flavorOriginal(drums);
  return `${opening} — ${flavor}. ${family.structOriginal}${sig}`
    .replace(/\s+/g, " ")
    .trim();
}

export const REMIX_TEMPLATES: RemixTemplate[] = (() => {
  const raw: RemixTemplate[] = [];
  // All lanes (21 verified-bank + expansion lanes) × 7 families × 2 jobs.
  for (const lane of ALL_LANES) {
    for (const fam of REMIX_FAMILIES_DEF) {
      for (const job of ["original", "preserve"] as const) {
        raw.push({
          key: `remix-lane-${lane}-${fam.id}-${job}`,
          lane,
          title: `${lane} — ${fam.label}`,
          source: `Remix · ${fam.label}`,
          job,
          family: fam.label,
          text: composeRemix(lane, lane, fam, job),
        });
      }
    }
  }
  // Strongest Ammo styles × 7 families × 2 jobs.
  for (const entry of STRONGEST_AMMO) {
    // Only include styles that actually exist in the Ammo bank.
    const exists = ammo.sections.some((sec) =>
      sec.styles.some((st) => st.name === entry.name),
    );
    if (!exists) continue;
    for (const fam of REMIX_FAMILIES_DEF) {
      for (const job of ["original", "preserve"] as const) {
        raw.push({
          key: `remix-ammo-${entry.name}-${fam.id}-${job}`,
          lane: entry.name,
          title: `${entry.name} — ${fam.label}`,
          source: `Remix · Ammo · ${fam.label}`,
          job,
          family: fam.label,
          text: composeRemix(entry.name, entry.drumsLane, fam, job),
        });
      }
    }
  }
  const seen = new Set<string>();
  return raw.filter((t) => {
    if (seen.has(t.text)) return false;
    seen.add(t.text);
    return true;
  });
})();

export function remixTemplatesFor(
  job?: JobType,
  lane?: string,
  family?: string,
): RemixTemplate[] {
  return REMIX_TEMPLATES.filter(
    (t) =>
      (job === undefined || t.job === job) &&
      (lane === undefined || t.lane === lane) &&
      (family === undefined || family === "all" || t.family === family),
  );
}

export function randomRemix(
  job?: JobType,
  lane?: string,
  family?: string,
): RemixTemplate | undefined {
  const pool = remixTemplatesFor(job, lane, family);
  if (pool.length === 0) return undefined;
  return pool[Math.floor(Math.random() * pool.length)];
}

/** Remix template pinned to an analyzed beat's measured lane — the
 * Faithful Enhance family by default, or a named family when given. */
export function remixFromAnalysis(
  a: { suggestedLane: string },
  job: JobType,
  family = "Faithful Enhance",
): RemixTemplate | undefined {
  return (
    REMIX_TEMPLATES.find(
      (t) => t.job === job && t.lane === a.suggestedLane && t.family === family,
    ) ?? randomRemix(job, a.suggestedLane)
  );
}

/**
 * Assemble a Simple package headed by a remix template instead of the
 * short directive: remix template + an inserted bank style prompt, as
 * one copy-ready paste. The lock line appears once on Preserve jobs.
 */
export function assembleRemixPackage(
  remixText: string,
  styleText: string,
  job: JobType,
): string {
  const head = remixText.trim().replace(/\s+/g, " ");
  let style = styleText.trim().replace(/\s+/g, " ");
  if (job === "preserve" && head.includes(LOCK_LINE)) {
    const prefix = `${LOCK_LINE}. `;
    if (style.startsWith(prefix)) style = style.slice(prefix.length);
    else if (style.startsWith(`${LOCK_LINE} — `))
      style = style.slice(LOCK_LINE.length + 3);
  }
  if (!style) return head;
  const styleSentence = /[.!?]$/.test(style) ? style : `${style}.`;
  return `${head} Inserted style: ${styleSentence}`.replace(/\s+/g, " ").trim();
}

/**
 * A full structural FRAMEWORK generated from a beat's measurements —
 * the measured lane's framework profile (section layout with bar
 * counts, builds / drops / breakdown placed per that lane's frame,
 * the full event family) fitted to the measured pocket, key, low end,
 * brightness and density. This is a Lyrics-map-shaped framework, not
 * a style prompt.
 *
 * PRESERVATION LAW (mandatory on the Preserve framework): it LEADS
 * with the exact lock line; every section directive in the underlying
 * map is tagged sample-untouched; all fills, builds, drops, rolls and
 * FX events are canvas-only (drums / 808s / extras) — never re-chop,
 * re-perform or restyle the sample or the sampled vocals. It pairs
 * with the Extend route. The Original framework may arrange freely.
 */
export function frameworkFromAnalysis(
  a: {
    suggestedLane: string;
    pocketLabel: string;
    keyName: string | null;
    lowShare: number;
    centroidHz: number;
    onsetRate: number;
    sampleWordsText?: string;
    sampleWordsKind?: "clear" | "fragment";
  },
  job: JobType,
): string {
  const fw = frameworkFor(a.suggestedLane);
  const keyPart = a.keyName ? `, ${a.keyName}` : "";
  // Header stays short on purpose: the full measured descriptors live
  // in the analysis style prompts, and the framework must stay inside
  // the 5,000-char Lyrics cap when loaded into Build.
  const measured = `${a.pocketLabel}${keyPart}`;
  let map = buildV6LyricsMap(a.suggestedLane, job);
  if (job === "preserve") {
    // The framework head already carries the lock line and the
    // canvas-only law, so the map's own lock parenthetical (its
    // second line) is dropped in this composition — that keeps a
    // framework carrying sample words inside the Lyrics cap. Every
    // section directive still carries its sample-untouched tag.
    map = map
      .split("\n")
      .filter((line) => !line.startsWith(`(${LOCK_LINE} —`))
      .join("\n");
  }
  // Sample content — only words actually entered as heard (never
  // invented or completed by the app), labeled clear or fragment.
  // The sampled vocals are part of the original recording and stay
  // preservation-locked regardless of what they say.
  //
  // Cap safety (measured defect): the sample line rides ON TOP of a
  // map that already runs to ~4,760 chars for the longest lane, so
  // long entered words pushed EVERY lane's framework over the
  // 5,000-char Lyrics cap (a 500-char entry measured 5,052–5,375
  // across all 41 lanes). The entry field caps at 120 chars, and
  // this composition trims the quoted words to fit as a last resort —
  // the trim is visible (an ellipsis inside the quote), never silent.
  const rawWords = (a.sampleWordsText ?? "").trim();
  const words = rawWords.slice(0, 120);
  const wordsTrimmed = rawWords.length > words.length;
  const shownWords = wordsTrimmed ? `${words}…` : words;
  const sampleLine = words
    ? a.sampleWordsKind === "fragment"
      ? `[Sample content — fragment, partial words only: "${shownWords}" — sampled vocals stay exactly as performed, preservation-locked]`
      : `[Sample content — sampled vocals, preservation-locked: "${shownWords}"]`
    : null;
  if (job === "preserve") {
    const head = [
      LOCK_LINE,
      `[Analysis framework: ${fw ? fw.name : "v6 detailed structure"} · ${a.suggestedLane} · ${measured}]`,
      `(Canvas-only events. Never re-chop, re-perform or restyle the sample or sampled vocals. Route: Extend.)`,
    ].join("\n");
    return sampleLine ? `${head}\n${sampleLine}\n${map}` : `${head}\n${map}`;
  }
  const head = [
    `[Analysis framework: ${fw ? fw.name : "v6 detailed structure"} · ${a.suggestedLane} · ${measured}]`,
    `(Arranged freely from the measurements — builds, drops and breakdown placed per this lane's framework.)`,
  ].join("\n");
  return sampleLine ? `${head}\n${sampleLine}\n${map}` : `${head}\n${map}`;
}

function lowDesc(lowShare: number): string {
  if (lowShare >= 0.45) return "heavy low end already in the pocket";
  if (lowShare >= 0.3) return "solid low end under the sample";
  return "lean low end, room for sub and 808 support";
}

function brightDesc(centroidHz: number): string {
  if (centroidHz >= 3800) return "bright top end, hats and air up front";
  if (centroidHz >= 2400) return "balanced top end";
  return "dark, dusty top end with rolled-off highs";
}

function densityDesc(onsetRate: number): string {
  if (onsetRate >= 3.5) return "busy hit density, drums stay active";
  if (onsetRate >= 1.8) return "steady hit density";
  return "sparse hit density with lots of open space";
}

/**
 * A Simple-box paragraph built from a beat's on-device measurements —
 * lane-pinned to the measured lane, exactly like the Build handoff.
 * Preserve carries the exact lock line inside the paragraph.
 */
export function simpleFromAnalysis(
  a: {
    suggestedLane: string;
    pocketLabel: string;
    keyName: string | null;
    lowShare: number;
    centroidHz: number;
    onsetRate: number;
    sampleWordsText?: string;
    sampleWordsKind?: "clear" | "fragment";
  },
  job: JobType,
): string {
  const keyPart = a.keyName ? `, ${a.keyName} tonal center` : "";
  const measured =
    `${a.pocketLabel}${keyPart}, ${lowDesc(a.lowShare)}, ` +
    `${brightDesc(a.centroidHz)}, ${densityDesc(a.onsetRate)}`;
  const drums = laneDrums(a.suggestedLane);
  const words = (a.sampleWordsText ?? "").trim();
  const sampleSentence = words
    ? a.sampleWordsKind === "fragment"
      ? ` Sample says (fragment, partial words only): "${words}".`
      : ` Sample says: "${words}".`
    : "";
  if (job === "preserve") {
    const head =
      `${LOCK_LINE}. Enhance this ${a.suggestedLane} upload — ${measured}.${sampleSentence} ` +
      `Drums, 808s and texture are the canvas only, spice only, sample stays lead. ` +
      `Drums and 808: ${drums}.`;
    return `${head} ${simpleStructureFor(a.suggestedLane, "preserve")}`;
  }
  const head =
    `${a.suggestedLane} instrumental — ${measured}, drums and 808s lead the arrangement.${sampleSentence} ` +
    `Drums and 808: ${drums}.`;
  return `${head} ${simpleStructureFor(a.suggestedLane, "original")}`;
}
